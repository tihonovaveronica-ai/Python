def vvc(name):
    '''ф-ция для ввода вещественного числа'''
    return float(input(f'{name}'))
def vt(name):
    '''ф-ция для ввода объема и темп. порции смеси'''
    print(f'порция смеси номер {name}')
    V=vvc('Введите объем порции смеси: ')
    T=vvc('введите температуру порции смеси: ')
    return [V,T]
def resv(v1,v2):
    '''нахождение итогового объема'''
    return v1+v2
def rest(v1,v2,t1,t2):
    '''нахождение итоговой температуры'''
    return (v1*t1+v2*t2)/(v1+v2)
def res(v1,v2,t1,t2):
    print('После объединения')
    print(f'{v1} литров воды температуры {t1}')
    print(f'и {v2} литров воды температуры {t2}')
    print(f'получим смесь {resv(v1,v2)} объема и {rest(v1,v2,t1,t2)} температуры.')
    
v1,t1=vt('1')
v2,t2=vt('2')
res(v1,v2,t1,t2)

from math import pi
from math import tan
def vc_ch(name):
    #ф-ция для ввода целого числа
    n=int(input(f'{name}'))
    return n

def vv_ch(name):
    #ф-ция для ввода вещественного числа
    n=float(input(f'{name}'))
    return n

def plos(n,a):
    #ф-ция для вычисления площади
    s=(n*a**2)/(4*tan(pi/n))
    return s

def res(a,n,s):
    #ф-цтя для вывода результата
    print(f'Длинна стороны многоугольника={a}')
    print(f'Количество сторон правильного многоугольника={n}')
    print(f'Площадь правильного многоугольника={s}')

a=vv_ch('Введите длинну стороны многоугольника: ')
n=vc_ch('Введите количество сторон многоугольника: ')
s=plos(n,a)
res(a,n,s)
